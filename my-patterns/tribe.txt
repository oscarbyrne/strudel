const tribe = {
  midiport: 'UM-ONE MIDI 1',
  midimap: {
    amp: 7,
    pitch: 80,
    glide: 81,
    edit: 82,
    cutoff: 74,
  },
  tonic: 'C4',
  sync() {
    return midicmd("clock*24*4").midi(this.midiport)
  },
  part(n) {
    const that = this
    return function(...patterns) {
      return stack(...patterns).midi({
        midiport: that.midiport,
        midichannel: n,
        midimap: 'tribe',
      })
    }
  },
  init(bpm) {
    setcpm(bpm/4)
    midimaps({ tribe: tribe.midimap })
  }
}

tribe.init(166)

let {glide, edit, pitch, cutoff} = createParams('glide', 'edit', 'pitch', 'cutoff')

const levels = register('levels', function(n, min, max, offset, pattern) {
  return pattern.div(n - 1).range(min, max).add(offset).div(127)
})

const lfo = function(signal, bars, min, max, offset) {
  return signal.slow(bars).seg(16 * bars).range(min, max).add(offset).div(127)
}

stack(
  // tribe.sync(),
  tribe.part(1)(note('C4').struct("[x ~ ~ ~] [~ ~ x ~] [~ ~ ~ ~] [~ ~ ~ ~]")),
  tribe.part(2)(note('C4').struct("[~ ~ ~ ~] [x ~ ~ ~] [~ ~ ~ ~] [x ~ ~ ~]")),
  tribe.part(3)(
    note('C4').struct(
      "[x x x x]!4"
    ),
    amp(
      "[7 2 3 4]!4".levels(8, 0, 100, 0)
    )
  ),
  tribe.part(5)(
    n(0).struct(
      "[~ ~ ~ ~ ] [x ~ x x] [x ~ x ~ ] [x ~ x x]"
    ).scale('E#:minor:blues').scaleTranspose(
      "[~ ~ ~ ~ ] [5 ~ 7 1] [0 ~ 1 ~ ] [5 ~ 5 1]"
    ).transpose(
      "[6P 5P 5P 6P]!4"
    ),
    amp("[0 1 0 2]!4".levels(3, 30, 60, 0)),
    glide("[0 1 1 0]!4".levels(2, 0, 32, 0)),
    edit(lfo(sine, 2, 0, 38, 64)),
    cutoff(lfo(sine.late(0.25), 8, 60, 100, 0))
  ),
)
